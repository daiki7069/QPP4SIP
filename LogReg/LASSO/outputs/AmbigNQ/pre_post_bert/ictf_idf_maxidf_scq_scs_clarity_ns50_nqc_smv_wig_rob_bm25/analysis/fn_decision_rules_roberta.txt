def should_use_qpp_correction(qpp_features, bert_prediction):
    """
    RoBERTaの予測をQPPで補正すべきかどうかを判定
    
    Args:
        qpp_features: dict, QPP特徴量の辞書
        bert_prediction: str, RoBERTaの予測 ('clarification' または 'answer')
    
    Returns:
        tuple: (should_correct: bool, correction_direction: str or None)
    """
    
    # ルール 1: WIG
    if qpp_features.get('WIG', float('-inf')) > 1.4816 and bert_prediction == 'answer':
        return True, 'answer'
    
    # ルール 2: nSigma
    if qpp_features.get('nSigma', float('-inf')) > 0.3460 and bert_prediction == 'answer':
        return True, 'answer'
    
    # ルール 3: NQC
    if qpp_features.get('NQC', float('-inf')) > 0.0876 and bert_prediction == 'answer':
        return True, 'answer'
    
    # ルール 4: Clarity
    if qpp_features.get('Clarity', float('-inf')) > 1.8695 and bert_prediction == 'answer':
        return True, 'answer'
    
    return False, None  # QPP補正不要
