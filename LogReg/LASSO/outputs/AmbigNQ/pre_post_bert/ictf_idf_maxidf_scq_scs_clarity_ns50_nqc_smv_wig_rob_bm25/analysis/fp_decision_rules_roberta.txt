def should_use_qpp_correction(qpp_features, bert_prediction):
    """
    RoBERTaの予測をQPPで補正すべきかどうかを判定
    
    Args:
        qpp_features: dict, QPP特徴量の辞書
        bert_prediction: str, RoBERTaの予測 ('clarification' または 'answer')
    
    Returns:
        tuple: (should_correct: bool, correction_direction: str or None)
    """
    
    # ルール 1: WIG
    if qpp_features.get('WIG', float('inf')) < 1.0786 and bert_prediction == 'clarification':
        return True, 'answer'
    
    # ルール 2: nSigma
    if qpp_features.get('nSigma', float('inf')) < 0.2549 and bert_prediction == 'clarification':
        return True, 'answer'
    
    # ルール 3: SClarity
    if qpp_features.get('SClarity', float('inf')) < 7.7283 and bert_prediction == 'clarification':
        return True, 'answer'
    
    # ルール 4: AvgIDF
    if qpp_features.get('AvgIDF', float('inf')) < 5.3187 and bert_prediction == 'clarification':
        return True, 'answer'
    
    return False, None  # QPP補正不要
