def should_use_qpp_correction(qpp_features, bert_prediction):
    """
    BERTの予測をQPPで補正すべきかどうかを判定
    
    Args:
        qpp_features: dict, QPP特徴量の辞書
        bert_prediction: str, BERTの予測 ('clarification' または 'answer')
    
    Returns:
        tuple: (should_correct: bool, correction_direction: str or None)
    """
    
    # ルール 1: WIG
    if qpp_features.get('WIG', float('-inf')) > 1.4417 and bert_prediction == 'answer':
        return True, 'answer'
    
    # ルール 2: SClarity
    if qpp_features.get('SClarity', float('-inf')) > 8.4574 and bert_prediction == 'answer':
        return True, 'answer'
    
    # ルール 3: AvgIDF
    if qpp_features.get('AvgIDF', float('-inf')) > 5.9373 and bert_prediction == 'answer':
        return True, 'answer'
    
    # ルール 4: nSigma
    if qpp_features.get('nSigma', float('-inf')) > 0.3293 and bert_prediction == 'answer':
        return True, 'answer'
    
    return False, None  # QPP補正不要
